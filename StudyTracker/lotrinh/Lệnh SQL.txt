Xem tất cả Ngày đã học (Thực tế):

sql
SELECT date AS "Ngày", subject AS "Môn", minutes AS "Số phút", notes AS "Ghi chú"
FROM days 
WHERE minutes > 0 OR subject != ''
ORDER BY date DESC;
2. Xem tất cả Deadline (Dù xong hay chưa xong):

sql
SELECT deadline AS "Ngày", title AS "Công việc", tag AS "Môn", 
       CASE WHEN done = 1 THEN '✅ Đã xong' ELSE '⏰ Chưa xong' END AS "Trạng thái"
FROM todos 
WHERE deadline IS NOT NULL AND deadline != ''
ORDER BY deadline DESC;
3. Xem tất cả các Phiên học (Pomodoro & Thủ công):

sql
SELECT date AS "Ngày", time(at) AS "Giờ", minutes AS "Phút", 
       tag AS "Môn", text AS "Ghi chú" 
FROM sessions 
ORDER BY date DESC, at DESC;
4. Xem tất cả các Môn học đã tạo:

sql
SELECT id, name AS "Tên Môn", color AS "Mã Màu", created_at AS "Ngày Tạo" 
FROM subjects;

sql

       completed_sessions AS "Đã xong"

1. Bảng tổng hợp TẤT CẢ Lịch Học (Kế hoạch + Thực tế + Deadline):
Lệnh này gom ngày nào có kế hoạch, ngày nào đã học, ngày nào có deadline ra chung 1 bảng cực dễ nhìn.

javascript
(function tongHopLichHoc() {
    const days = window.DB.query('SELECT * FROM days');
    const todos = window.DB.query('SELECT * FROM todos');
    const ketQua = {};

    // Quét Kế hoạch và Thực tế
    days.forEach(d => {
        let slots = [];
        try { slots = JSON.parse(d.slots_json || '[]'); } catch(e){}
        if (slots.length > 0 || d.minutes > 0 || d.subject) {
            ketQua[d.date] = {
                "Ngày": d.date,
                "Môn": d.subject || '---',
                "Phút đã học": d.minutes || 0,
                "Kế hoạch (Khung giờ)": slots.map(s => `${s.hour}:00 - ${s.text}`).join(' | ') || 'Trống',
                "Deadline": []
            };
        }
    });

    // Quét Deadline
    todos.forEach(t => {
        if (t.deadline) {
            if (!ketQua[t.deadline]) {
                ketQua[t.deadline] = {
                    "Ngày": t.deadline, "Môn": '---', "Phút đã học": 0, "Kế hoạch (Khung giờ)": 'Trống', "Deadline": []
                };
            }
            ketQua[t.deadline]["Deadline"].push(`[${t.done ? 'Xong' : 'Chưa'}] ${t.title} (${t.tag || 'Ko môn'})`);
        }
    });

    const mang = Object.values(ketQua).sort((a, b) => b["Ngày"].localeCompare(a["Ngày"]));
    mang.forEach(item => { item["Deadline"] = item["Deadline"].join(' | ') || 'Trống'; });
    
    console.log("%c=== TỔNG HỢP LỊCH HỌC ===", "color: #4f46e5; font-weight: bold; font-size: 14px");
    console.table(mang);
})();
